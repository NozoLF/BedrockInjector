cmake_minimum_required(VERSION 3.21)
project(BedrockInProcess LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(Injector WIN32 src/injector.cpp)
target_compile_definitions(Injector PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(Injector PRIVATE user32 comdlg32)
add_library(BedrockMenu SHARED src/menu.cpp)
target_compile_definitions(BedrockMenu PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(BedrockMenu PRIVATE user32 gdi32)
set_target_properties(Injector BedrockMenu PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Release")
